-- Q12 [서브쿼리] 전체 수업 평균 수강료보다 비싼 수업을 조회한다 (스칼라 서브쿼리)
SELECT id, title, category, price
FROM lesson
WHERE price > (SELECT AVG(price) FROM lesson)
ORDER BY price DESC, id;
