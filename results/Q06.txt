-- Q06 [INNER JOIN] 수업별 담당 트레이너와 그 전문 분야를 조회한다
SELECT l.id AS lesson_id, l.title, l.category, t.name AS trainer_name, t.specialty
FROM lesson l
INNER JOIN trainer t ON t.id = l.trainer_id
ORDER BY l.id;
