-- Q13 [서브쿼리] 예약 기록이 한 건도 없는 회원을 찾는다 (NOT EXISTS)
SELECT m.id, m.name, m.email, m.joined_at
FROM member m
WHERE NOT EXISTS (SELECT 1 FROM booking b WHERE b.member_id = m.id)
ORDER BY m.id;
