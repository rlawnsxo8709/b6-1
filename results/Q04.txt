-- Q04 [기본조회] 정원 15명 이상이면서 수강료 2만 원 이하인 부담 없는 수업을 조회한다
SELECT id, title, category, capacity, price
FROM lesson
WHERE capacity >= 15 AND price <= 20000
ORDER BY price, capacity DESC;
