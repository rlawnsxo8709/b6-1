-- Q08 [LEFT JOIN] 예약이 한 건도 없는 수업을 찾는다 (IS NULL 로 짝이 없는 행만 남김)
SELECT l.id AS lesson_id, l.title, l.starts_at
FROM lesson l
LEFT JOIN booking b ON b.lesson_id = l.id
WHERE b.id IS NULL
ORDER BY l.starts_at;
