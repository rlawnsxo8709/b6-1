-- Q02 [기본조회] 수강료가 높은 상위 5개 수업을 조회한다 (같은 가격이면 빠른 수업 먼저)
-- LIMIT 은 SQLite·MySQL·PostgreSQL 이 공통으로 지원하는 문법이다.
SELECT id, title, category, price
FROM lesson
ORDER BY price DESC, starts_at
LIMIT 5;
